# Sair: Ctrl+B depois D
# Voltar: tmux attach -t ngrok

screen -S ngrok
ngrok http 3000
# Sair: Ctrl+A depois D
# Voltar: screen -r ngrok

# Com nohup:
nohup ngrok http 3000 > ngrok.log 2>&1 &

# Pegar URL e copiar (se tiver xclip):
wget -qO- http://localhost:4040/api/tunnels | python3 -c "import sys, json; url=[t['public_url'] for t in json.load(sys.stdin)['tunnels']][0] if json.load(sys.stdin).get('tunnels') else ''; print(url)" | xclip -selection clipboard

# Gerar QR Code para escanear no celular (se tiver qrencode):
sudo apt install qrencode
wget -qO- http://localhost:4040/api/tunnels | python3 -c "import sys, json; print([t['public_url'] for t in json.load(sys.stdin)['tunnels']][0])" | qrencode -t UTF8
